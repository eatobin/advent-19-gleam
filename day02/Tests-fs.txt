module Tests

open System
open Xunit
open FsUnit.Xunit
open Library.Intcode
open FSharpx.Collections

[<Fact>]
let ``a test all by itself`` () = 78 |> should equal 78

// Part A

let instruction1 = Map [ ('a', 0); ('b', 0); ('c', 0); ('d', 0); ('e', 6) ]
let instruction3 = Map [ ('a', 0); ('b', 0); ('c', 4); ('d', 5); ('e', 6) ]
let instruction5 = Map [ ('a', 2); ('b', 3); ('c', 4); ('d', 5); ('e', 6) ]
let memoryAsCSVString = "10,11,1"
let thisMemory: PersistentVector<int> = PersistentVector.ofSeq [ 10; 11; 1 ]

let intCode =
    { pointer = 0
      memory = thisMemory
      actions = [] }

let thisMemoryX: PersistentVector<int> = PersistentVector.ofSeq [ 0; 2; 1; 0 ]

let intcodeX =
    { pointer = 0
      memory = thisMemoryX
      actions = [] }

let thisMemoryAddMult: PersistentVector<int> = PersistentVector.ofSeq [ 0; 2; 1; 0 ]
let aocMemory1 = "1,0,0,3,99"
let aocMemory2 = "1,9,10,3,2,3,11,0,99,30,40,50"
let aocMemory3 = "1,0,0,0,99"
let aocMemory4 = "2,3,0,3,99"
let aocMemory5 = "2,4,4,5,99,0"
let aocMemory6 = "1,1,1,4,99,5,6,0,99"

let intCodeAddMultExit =
    { pointer = 0
      memory = thisMemoryAddMult
      actions = [] }

let intCodeAdd =
    { pointer = 4
      memory = PersistentVector.ofSeq [ 3; 2; 1; 0 ]
      actions = [ Add ] }

let intCodeMult =
    { pointer = 4
      memory = PersistentVector.ofSeq [ 2; 2; 1; 0 ]
      actions = [ Multiply ] }

let intCodeExit =
    { pointer = 0
      memory = thisMemoryAddMult
      actions = [ Exit ] }

[<Fact>]
let ``make an Instruction 1`` () =
    makeInstruction 6 |> should equal instruction1

[<Fact>]
let ``make an Instruction 3`` () =
    makeInstruction 456 |> should equal instruction3

[<Fact>]
let ``make an Instruction 5`` () =
    makeInstruction 23456 |> should equal instruction5

[<Fact>]
let ``make an Memory`` () =
    makeMemory memoryAsCSVString |> should equal thisMemory

[<Fact>]
let ``lookup a valid Memory index - pw`` () = pw intCode 2 |> should equal 1

[<Fact>]
let ``lookup a valid Memory index - pr`` () = pr intCode 2 |> should equal 11

[<Fact>]
let ``lookup an invalid Memory index`` () =
    let myException =
        Assert.Throws<IndexOutOfRangeException>(fun () ->
            if pw intCode 33 <> 0 then
                raise (IndexOutOfRangeException()) |> ignore)

    Assert.Contains("Index was outside the bounds of the array.", myException.Message)

[<Fact>]
let ``lookup a valid aParam`` () =
    aParam instruction1 intcodeX |> should equal 0

[<Fact>]
let ``lookup a valid bParam`` () =
    bParam instruction1 intcodeX |> should equal 2

[<Fact>]
let ``lookup a valid cParam`` () =
    cParam instruction1 intcodeX |> should equal 1

[<Fact>]
let ``1 plus 2 should be set at 0 and pointer should be 4`` () =
    add instruction1 intCodeAddMultExit |> should equal intCodeAdd

[<Fact>]
let ``1 times 2 should be set at 0 and pointer should be 4`` () =
    multiply instruction1 intCodeAddMultExit |> should equal intCodeMult
