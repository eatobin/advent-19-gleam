namespace Library

open System.Text
open FSharpx.Collections

module Intcode =

    // Instruction:
    // ABCDE
    // 01234
    // 01002
    // 34(DE) - two-digit opcode,      02 == opcode 2
    //  2(C) - mode of 1st parameter,  0 == position mode
    //  1(B) - mode of 2nd parameter,  1 == immediate mode
    //  0(A) - mode of 3rd parameter,  0 == position mode,
    //                                   omitted due to being a leading zero
    // 0 1 or 2 = left-to-right position after 2 digit opcode
    // p i or r = position, immediate or relative mode
    // r or w = read or write

    type Instruction = Map<char, int>
    type Pointer = int
    type Key = int
    type Value = int
    type Memory = PersistentVector<int>
    type MemoryAsCSVString = string
    type PointerOffset = int

    type IntCodeAction =
        | Add
        | Multiply
        | Exit

    type Actions = IntCodeAction list

    type IntCode =
        { pointer: Pointer
          memory: Memory
          actions: Actions }

    let pointerOffsetC: int = 1
    let pointerOffsetB: int = 2
    let pointerOffsetA: int = 3

    let charToInt (aChar: byte) : int = int (aChar - 48uy)

    let makeInstruction (op: int) : Instruction =
        let values = Encoding.ASCII.GetBytes $"%05i{op}" |> Array.map charToInt

        let keys = [| 'a'; 'b'; 'c'; 'd'; 'e' |]
        Array.zip keys values |> Map.ofArray

    let makeMemory (memoryAsCSVString: MemoryAsCSVString) : Memory =
        memoryAsCSVString.Split(",") |> Array.map int |> PersistentVector.ofSeq

    let updatedMemory (noun: int) (verb: int) (memory: Memory) : Memory =
        memory |> PersistentVector.update 1 noun |> PersistentVector.update 2 verb

    let keyToKey (intCode: IntCode) (pointerOffsetParam: PointerOffset) : Key =
        intCode.memory[(intCode.pointer + pointerOffsetParam)]

    let pw (intCode: IntCode) (pointerOffsetParam: PointerOffset) : Key = keyToKey intCode pointerOffsetParam

    let pr (intCode: IntCode) (pointerOffsetParam: PointerOffset) : Value =
        intCode.memory[keyToKey intCode pointerOffsetParam]

    let aParam (instruction: Instruction) (intCode: IntCode) : int =
        match instruction['a'] with
        | 0 -> pw intCode pointerOffsetA // a-p-w
        | _ -> raise (System.ArgumentException("Instruction is not valid"))

    let bParam (instruction: Instruction) (intCode: IntCode) : int =
        match instruction['b'] with
        | 0 -> pr intCode pointerOffsetB // b-p-r
        | _ -> raise (System.ArgumentException("Instruction is not valid"))

    let cParam (instruction: Instruction) (intCode: IntCode) : int =
        match instruction['c'] with
        | 0 -> pr intCode pointerOffsetC // c-p-r
        | _ -> raise (System.ArgumentException("Instruction is not valid"))

    let add (instruction: Instruction) (intCode: IntCode) : IntCode =
        { pointer = intCode.pointer + 4
          memory =
            intCode.memory
            |> PersistentVector.update
                (aParam instruction intCode)
                ((cParam instruction intCode) + (bParam instruction intCode))
          actions = Add :: intCode.actions }

    let multiply (instruction: Instruction) (intCode: IntCode) : IntCode =
        { pointer = intCode.pointer + 4
          memory =
            intCode.memory
            |> PersistentVector.update
                (aParam instruction intCode)
                ((cParam instruction intCode) * (bParam instruction intCode))
          actions = Multiply :: intCode.actions }

    let exit (intCode: IntCode) : IntCode =
        { pointer = intCode.pointer
          memory = intCode.memory
          actions = Exit :: intCode.actions}

    [<TailCall>]
    let rec runOpCode (intCode: IntCode) : IntCode =
        let instruction: Instruction = makeInstruction intCode.memory[intCode.pointer]

        match instruction['e'] with
        | 1 -> runOpCode (add instruction intCode)
        | 2 -> runOpCode (multiply instruction intCode)
        | 9 -> exit intCode
        | _ -> raise (System.ArgumentException("Instruction is not valid"))
