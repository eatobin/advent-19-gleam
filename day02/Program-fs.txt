namespace App

open Library.Intcode

module Program =
    do
        let memoryAsCSVString: string =
            "1,0,0,3,1,1,2,3,1,3,4,3,1,5,0,3,2,10,1,19,2,9,19,23,2,13,23,27,1,6,27,31,2,6,31,35,2,13,35,39,1,39,10,43,2,43,13,47,1,9,47,51,1,51,13,55,1,55,13,59,2,59,13,63,1,63,6,67,2,6,67,71,1,5,71,75,2,6,75,79,1,5,79,83,2,83,6,87,1,5,87,91,1,6,91,95,2,95,6,99,1,5,99,103,1,6,103,107,1,107,2,111,1,111,5,0,99,2,14,0,0"

        let firstMemory: Memory = makeMemory memoryAsCSVString

        let initialState: IntCode =
            { pointer = 0
              memory = updatedMemory 12 2 firstMemory
              actions = [] }

        // part A
        let finalStateA: IntCode = runOpCode initialState
        let answer1: int = finalStateA.memory[0]

        printfn $"\nPart A: %i{answer1}, correct: 2890696"
        printfn $"%A{List.rev finalStateA.actions}"

        // part B
        let answer2: (int * IntCode) =
            [ for noun: int in 0..99 do
                  for verb: int in 0..99 do
                      let candidateIntCode: IntCode =
                          (runOpCode
                              { pointer = 0
                                memory = updatedMemory noun verb firstMemory
                                actions = [] })

                      let candidate: int = candidateIntCode.memory[0]

                      if (candidate = 19_690_720) then
                          yield ((100 * noun) + verb, candidateIntCode) ]
                .Head

        printfn $"\nPart B: %i{fst answer2}, correct: 8226"
        printfn $"%A{List.rev (snd answer2).actions}"
